cmake_minimum_required(VERSION 3.8)
project(fsd_rviz_plugins)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rviz_common REQUIRED)
find_package(rviz_default_plugins REQUIRED)
find_package(pluginlib REQUIRED)
find_package(message_filters REQUIRED)
find_package(std_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_library(fsd_safe_image SHARED src/safe_image.cpp src/diagnostics_panel.cpp)
ament_target_dependencies(fsd_safe_image rviz_common rviz_default_plugins pluginlib message_filters std_msgs rclcpp)
target_link_libraries(fsd_safe_image Qt5::Widgets)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS fsd_safe_image LIBRARY DESTINATION lib)
ament_package()
